#include <metal_stdlib>
using namespace metal;

struct VividUniforms { float assembly; float pixels; };
struct VividVertex { float4 position [[position]]; float2 uv; };
struct VividStar { float4 homeScatter; float4 angle; };
struct VividStarVertex {
    float4 position [[position]];
    float size [[point_size]];
    float strength;
};

vertex VividVertex vividMarkVertex(uint id [[vertex_id]]) {
    const float2 positions[] = {float2(-1,-1), float2(1,-1), float2(-1,1), float2(1,1)};
    float2 p = positions[id];
    return {float4(p,0,1), p * .5 + .5};
}

fragment float4 vividMarkFragment(VividVertex in [[stage_in]],
                                 texture2d<float> mark [[texture(0)]],
                                 constant VividUniforms &u [[buffer(0)]]) {
    constexpr sampler s(coord::normalized, address::clamp_to_edge, filter::linear);
    float2 logoUV = (in.uv - .5) * 1.5 + .5;
    if (any(logoUV < 0.) || any(logoUV > 1.)) return float4(0.);
    float4 original = mark.sample(s, float2(logoUV.x, 1. - logoUV.y));
    float2 p = logoUV - float2(.5,.51);
    float radius = length(p);
    float reflection = (p.x * .75 + p.y) * 7. + .4;
    float shine = pow(max(0.,sin(reflection)),10.);
    float soft = .5 + .5 * sin(reflection - 1.);
    float2 offset = float2(.004,.006);
    float2 a = logoUV + offset, b = logoUV - offset;
    float bevel = clamp((1. - mark.sample(s,float2(a.x,1.-a.y)).a) +
                        (1. - mark.sample(s,float2(b.x,1.-b.y)).a),0.,1.);
    float material = dot(original.rgb,float3(.333));
    float metal = .13 + material * .7 + soft * .18 + shine * .5 + bevel * .48;
    float grain = fract(sin(dot(floor(logoUV * 180.),float2(12.9898,78.233))) * 43758.5453);
    float arrival = radius * .3 + grain * .14;
    float reveal = smoothstep(.40 + arrival,.62 + arrival,u.assembly);
    float wash = sin(clamp((u.assembly - .46)/.54,0.,1.) * 3.14159) * .5;
    float opacity = original.a * reveal;
    return float4(float3(metal + wash) * float3(.975,.985,1.) * opacity, opacity);
}

vertex VividStarVertex vividStarVertex(uint id [[vertex_id]],
                                      const device VividStar *stars [[buffer(0)]],
                                      constant VividUniforms &u [[buffer(1)]]) {
    VividStar star = stars[id];
    float loose = 1. - u.assembly;
    float angle = star.angle.x + loose * 1.8;
    float2 orbit = float2(cos(angle),sin(angle));
    float2 position = star.homeScatter.xy + orbit * star.homeScatter.z * loose;
    float size = min(22.,(1.5 + loose * star.homeScatter.w * 9.) * u.pixels / 640.);
    float strength = (1. - smoothstep(.72,1.,u.assembly)) * (.45 + .4 * star.homeScatter.w);
    return {float4(position/1.5,0.,1.), size, strength};
}

fragment float4 vividStarFragment(VividStarVertex in [[stage_in]], float2 point [[point_coord]]) {
    float r = length(point - .5) * 2.;
    float glow = exp(-r*r*5.) * (1. - smoothstep(.65,1.,r)) * in.strength;
    return float4(float3(.94,.97,1.) * glow,glow);
}
